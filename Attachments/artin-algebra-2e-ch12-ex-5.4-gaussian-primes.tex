\documentclass[tikz,border=6pt]{standalone}
\usepackage{amsmath,amssymb}
\begin{document}
\begin{tikzpicture}[scale=0.38,>=stealth]
  % Mathematical range: all Gaussian integers a+bi with a^2+b^2 <= 100.
  \draw[->,thin] (-10.7,0) -- (10.7,0) node[right] {$\operatorname{Re}$};
  \draw[->,thin] (0,-10.7) -- (0,10.7) node[above] {$\operatorname{Im}$};
  \draw[dashed,gray] (0,0) circle[radius=10];

  \foreach \t in {-10,-5,5,10} {
    \draw[thin] (\t,0.13) -- (\t,-0.13) node[below=2pt] {\scriptsize $\t$};
    \draw[thin] (0.13,\t) -- (-0.13,\t) node[left=2pt] {\scriptsize $\t$};
  }

  % Background lattice points.
  \foreach \a in {-10,...,10} {
    \foreach \b in {-10,...,10} {
      \pgfmathtruncatemacro{\N}{\a*\a+\b*\b}
      \ifnum\N<101\relax
        \fill[gray!50] (\a,\b) circle[radius=0.75pt];
      \fi
    }
  }

  % Gaussian primes: off-axis prime norm, or axis primes p == 3 mod 4.
  \foreach \a in {-10,...,10} {
    \foreach \b in {-10,...,10} {
      \pgfmathtruncatemacro{\N}{\a*\a+\b*\b}
      \pgfmathtruncatemacro{\isprime}{
        ((abs(\a)>0)&&(abs(\b)>0)&&
          ((\N==2)||(\N==5)||(\N==13)||(\N==17)||
           (\N==29)||(\N==37)||(\N==41)||(\N==53)||
           (\N==61)||(\N==73)||(\N==89)||(\N==97)))
        ||
        (((abs(\a)==0)||(abs(\b)==0))&&
          (((abs(\a)+abs(\b))==3)||((abs(\a)+abs(\b))==7)))
      }
      \ifnum\isprime=1\relax
        \fill[black] (\a,\b) circle[radius=1.55pt];
      \fi
    }
  }

  \fill[black] (-9.8,-11.5) circle[radius=1.55pt]
    node[right=4pt] {\small Gaussian prime};
  \fill[gray!50] (1.3,-11.5) circle[radius=0.75pt]
    node[black,right=4pt] {\small Gaussian integer};
\end{tikzpicture}
\end{document}
